# 树套树

树套树就是在线段上上套线段树，比较典型的可以求三维偏序。可以排序先确保一维，然后用线段树维护一维，线段树每个结点再长出一棵线段树维护第二位。
一般来说第二棵线段树需要动态开点，否则空间不够。如果第一棵需要维护的范围比较大，而又不能离线离散化，那也需要做成动态开点。
以陌上花开为例：
查询：先查 $x$，对于合适的 $x$ 范围，进行 $y$ 范围的查询。
修改：先定位 $x$，对于合适的 $x$ 范围，对 $y$ 范围的查询。

第一棵树每个线段对应于一段 $x$ 坐标，查询时如果 $x$ 盖住这个范围内，那么这个范围的所有 $y$ 都符合查询要求，直接查就行了，否则要递归到更小的区间。
第一棵树每个线段对应于一段 $x$ 坐标，修改时如果 $x$ 在这个范围内，那么就需要对这个范围内的第二棵树进行修改。修改完后，判断其属于左右，相应修改左右子树。
```cpp
int queryy(int node, int l, int r, int y) {
    if (!node) return 0;
    if (y >= r) return seg[node].info;
    int mid = l + r >> 1;
    int ret = queryy(seg[node].l, l, mid, y);
    if (y > mid) ret += queryy(seg[node].r, mid + 1, r, y);
    return ret;
}
int queryx(int node, int l, int r, int x, int y) {
    if (x >= r) return queryy(root[node], 1, m, y);
    int mid = l + r >> 1;
    int ret = queryx(node << 1, l, mid, x, y);
    if (x > mid) ret += queryx(node << 1 | 1, mid + 1, r, x, y);
    return ret;
}
void pushup(int node) { seg[node].info = seg[seg[node].l].info + seg[seg[node].r].info; }
void updatey(int& node, int l, int r, int y, int v) {
    if (!node) node = ++tot;
    if (l == r) {
        seg[node].info += v;
        return;
    }
    int mid = l + r >> 1;
    if (y <= mid)
        updatey(seg[node].l, l, mid, y, v);
    else
        updatey(seg[node].r, mid + 1, r, y, v);
    pushup(node);
}
void updatex(int node, int l, int r, int x, int y, int v) {
    updatey(root[node], 1, m, y, v);
    if (l == r) return;
    int mid = l + r >> 1;
    if (x <= mid)
        updatex(node << 1, l, mid, x, y, v);
    else
        updatex(node << 1 | 1, mid + 1, r, x, y, v);
}
```
