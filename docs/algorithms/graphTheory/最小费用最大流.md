基本操作就是反复做下面两件事：
- 用 spfa 找 $S \rightarrow T$ 的最低代价。
- 按该路径瓶颈流量增广。

直到无法找到 $S \rightarrow T$ 的通路。
```cpp
struct MCMF {
    int head[N], edge[N], n, tot;
    int to[M], nxt[M], src, dst;
    ll dis[N], len[M], cap[M], flow[N], maxflow, cost;
    bool in[N];
    MCMF(int _n) : n(_n) { fill(head, head + n, -1); }
    void add(int u, int v, ll f, ll c) {
        to[tot] = v, nxt[tot] = head[u], cap[tot] = f, len[tot] = c, head[u] = tot++;
        to[tot] = u, nxt[tot] = head[v], cap[tot] = 0, len[tot] = -c, head[v] = tot++;
    }
    bool spfa() {
        deque<int> dq;
        fill(dis, dis + n, LLONG_MAX), fill(in, in + n, false);
        dis[src] = 0, flow[src] = LLONG_MAX;
        dq.push_back(src), in[src] = true;
        while (!dq.empty()) {
            int u = dq.front(), v;
            dq.pop_front();
            for (int i = head[u]; ~i; i = nxt[i])
                if (dis[v = to[i]] > dis[u] + len[i] && cap[i]) {
                    dis[v] = dis[u] + len[i];
                    edge[v] = i;
                    flow[v] = min(flow[u], cap[i]);
                    if (!in[v]) {
                        in[v] = true;
                        if (dq.empty() || dis[v] >= dis[dq.front()])
                            dq.push_back(v);
                        else
                            dq.push_front(v);
                    }
                }
            in[u] = false;
        }
        return dis[dst] != LLONG_MAX;
    }
    void aug() {
        ll fl = flow[dst];
        int now = dst;
        maxflow += fl, cost += fl * dis[dst];
        while (now != src) {
            int eg = edge[now];
            cap[eg] -= fl, cap[eg ^ 1] += fl;
            now = to[eg ^ 1];
        }
    }
    void mcmf(int _src, int _dst) {
        src = _src, dst = _dst;
        maxflow = cost = 0;
        while (spfa()) aug();
    }
    void clear() { fill(head, head + n, -1), tot = 0; }
};
```
这里做 spfa 用到了一个优化，如果当前结点权值比队头小，则放入队头，否则放入队尾。
单次 spfa 最坏复杂度为 $O(VE)$，总流量为 $F$，则总复杂度为 $O(VEF)$。
如果非可以设计的流量和刻意设计的网络结构，则实际复杂度远远达不到。

如果边长都非常小，则增广时参考 Dinic 可以多路增广，则常数更小一些，但是如果边权很随意，则很难出现多条路的总权值一样，即没有效果，常数反而差。
```cpp
struct MCMF {
    int head[N], cur[N], n, tot;
    int to[M], nxt[M], src, dst;
    ll cap[M], len[M], dis[N], cost, maxflow;
    bool in[N], vis[N];
    MCMF(int _n) : n(_n) { tot = 0, fill(head, head + n, -1); }
    void add(int u, int v, ll f, ll c) {
        to[tot] = v, nxt[tot] = head[u], cap[tot] = f, len[tot] = c, head[u] = tot++;
        to[tot] = u, nxt[tot] = head[v], cap[tot] = 0, len[tot] = -c, head[v] = tot++;
    }
    bool spfa() {
        deque<int> dq;
        fill(dis, dis + n, LLONG_MAX), copy(head, head + n, cur), fill(vis, vis + n, false);
        fill(in, in + n, false), in[src] = true;
        dq.push_back(src), dis[src] = 0;
        while (!dq.empty()) {
            int u = dq.front(), v;
            dq.pop_front();
            for (int i = head[u]; ~i; i = nxt[i])
                if (dis[v = to[i]] > dis[u] + len[i] && cap[i]) {
                    dis[v] = dis[u] + len[i];
                    if (!in[v]) {
                        in[v] = true;
                        if (dq.empty() || dis[v] >= dis[dq.front()])
                            dq.push_back(v);
                        else
                            dq.push_front(v);
                    }
                }
            in[u] = false;
        }
        return dis[dst] != LLONG_MAX;
    }
    ll dfs(int now, ll maxflow) {
        if (now == dst) {
            cost += maxflow * dis[now];
            return maxflow;
        }
        vis[now] = true;
        int v;
        ll ret = 0;
        for (int& i = cur[now]; ~i; i = nxt[i])
            if (cap[i] && !vis[v = to[i]] && dis[v] == dis[now] + len[i]) {
                ll flow = dfs(v, min(maxflow, cap[i]));
                cap[i] -= flow, cap[i ^ 1] += flow;
                ret += flow, maxflow -= flow;
                if (!maxflow) break;
            }
        vis[now] = false;
        return ret;
    }
    void run(int _src, int _dst) {
        src = _src, dst = _dst;
        maxflow = cost = 0;
        while (spfa()) maxflow += dfs(src, LLONG_MAX);
    }
    void clear() { fill(head, head + n, -1), tot = 0; }
};
```

可以运用 Johnson 求最短路的方法先用 spfa 求出点权，然后在调权的图上用 dijkstra 求最短路。
SPFA 求最短路的复杂度 $O(VE)$，之后增广 $O(F)$ 轮，每轮复杂度 $O(ElogV)$，总体复杂度 $O(VE+FElogV)$。
```cpp
struct MCMF {
    int head[N], n, tot;
    int to[M], nxt[M], edge[N], src, dst;
    ll cap[M], len[M], dis[N], h[N], flow[N], cost, maxflow;
    bool in[N];
    MCMF(int _n) : n(_n) { tot = 0, fill(head, head + n, -1); }
    void add(int u, int v, ll f, ll c) {
        to[tot] = v, nxt[tot] = head[u], cap[tot] = f, len[tot] = c, head[u] = tot++;
        to[tot] = u, nxt[tot] = head[v], cap[tot] = 0, len[tot] = -c, head[v] = tot++;
    }
    void spfa() {
        deque<int> dq;
        fill(h, h + n, 0), fill(in, in + n, false);
        dq.push_back(src), in[src] = true;
        while (!dq.empty()) {
            int u = dq.front(), v;
            dq.pop_front(), in[u] = false;
            for (int i = head[u]; ~i; i = nxt[i])
                if (h[v = to[i]] > h[u] + len[i] && cap[i]) {
                    h[v] = h[u] + len[i];
                    if (!in[v]) {
                        in[v] = true;
                        if (dq.empty() || h[v] >= h[dq.front()])
                            dq.push_back(v);
                        else
                            dq.push_front(v);
                    }
                }
        }
    }
    bool dij() {
        priority_queue<pli, vector<pli>, greater<>> pq;
        flow[src] = LLONG_MAX, fill(dis, dis + n, inf);
        pq.emplace(0, src), dis[src] = 0;
        int v;
        while (!pq.empty()) {
            auto [len0, now] = pq.top();
            pq.pop();
            if (dis[now] != len0) continue;
            for (int i = head[now]; ~i; i = nxt[i]) {
                v = to[i];
                if (cap[i] && len0 + len[i] + h[now] - h[v] < dis[v]) {
                    dis[v] = len0 + len[i] + h[now] - h[v];
                    flow[v] = min(flow[now], cap[i]), edge[v] = i;
                    pq.emplace(dis[v], v);
                }
            }
        }
        return dis[dst] != inf;
    }
    void aug() {
        ll fl = flow[dst];
        int now = dst;
        maxflow += fl, cost += fl * (dis[dst] + h[dst] - h[src]);
        while (now != src) {
            int eg = edge[now];
            cap[eg] -= fl, cap[eg ^ 1] += fl;
            now = to[eg ^ 1];
        }
    }
    void run(int _src, int _dst) {
        src = _src, dst = _dst;
        maxflow = cost = 0;
        spfa();
        while (dij()) {
            aug();
            for (int i = 0; i < n; i++) h[i] += dis[i];
        }
    }
    void clear() { fill(head, head + n, -1), tot = 0; }
};
```
注意，每次增广之后，由于有些边会满流，导致 $S \rightarrow T$ 的最短路发生变化，需要相应调权。
设 $h_i$ 是原来从 $S$ 到 $V_i$ 的实际距离。当跑一遍 dijkstra 后，得到各点在调权图上的距离 $dis_i'$，而真实的距离 $dis_i=dis_i'-h_s+h_i=dis_i'+h_i$。
由于每一轮调权其实都是把实际距离作为权重，实际的 $dis_i=dis_i'+h_i$，所以新的权重即为 $dis_i'+h_i$。
这里注意，虽然有部分点不可达，但是可以同样操作，因为他们的 $cap$ 为 $0$，调权之后 $cap$ 还是为零。
